%% PART 1 — OLS SPECIFICATIONS
%% ====================================================================

%% ---------- Slide: The baseline problem ----------
\begin{frame}{Covariates fix broken parallel trends --- but the spec has to be right}
\vspace{0.4em}
The standard TWFE with additive $X$:
\[
  Y_{it} \;=\; \alpha_i + \lambda_t + \delta\,(D_i \times \text{Post}_t) + \boldsymbol{\beta} X_i + \varepsilon_{it}
\]
This imposes \highlight{two strong hidden assumptions}:
\vspace{0.3em}
\begin{enumerate}\setlength\itemsep{0.35em}
  \item The slope of $X$ on $Y(0)$ is the \textbf{same in pre and post}. (No time-varying returns.)
  \item The treatment effect $\delta$ is \textbf{the same for every value of $X$}. (Homogeneous TE in $X$.)
\end{enumerate}
\vspace{0.4em}
\meta{If either assumption fails, adding $X$ additively gives you the wrong answer. Specification is not a cosmetic choice.}
\end{frame}

%% ---------- Slide: Worry 1 — time-varying returns ----------
\begin{frame}{Worry 1: the effect of $X$ on $Y(0)$ changes over time}
\vspace{0.3em}
\textbf{Problem.} \;$\beta$ in the additive model is a single number. But returns to education, age, race gap etc.\ can shift at the same moment as your policy. That shift looks exactly like a treatment effect.

\vspace{0.4em}
\textbf{Fix: interact $\text{Post}$ with $X$.}
\[
  Y_{it} \;=\; \alpha_i + \lambda_t
        \;+\; \delta\,(D_i \times \text{Post}_t)
        \;+\; \boldsymbol{\beta}\,X_i
        \;+\; \boldsymbol{\gamma}\,(\text{Post}_t \times X_i)
        \;+\; \varepsilon_{it}
\]
$\boldsymbol{\gamma}$ picks up the \textit{change} in the slope of $X$ between pre and post. \\[0.3em]
The ATT estimate $\hat\delta$ is now protected against returns-to-$X$ shifting at the policy break.

\vspace{0.3em}
\meta{This is equivalent to interacting every covariate with a time dummy. Cost: $(k+1)$ extra parameters for $k$ covariates.}
\end{frame}

%% ---------- Slide: Worry 2 — heterogeneous TE in X ----------
\begin{frame}{Worry 2: the treatment effect itself varies with $X$}
\vspace{0.3em}
\textbf{Problem.} \;If $\delta_i = \delta(X_i)$, the additive spec estimates a \textit{variance-weighted} average of $\delta(X)$ --- not the ATT. The weights depend on $X$'s variation across the sample, not on the share treated.

\vspace{0.4em}
\textbf{Fix: interact $D$ (the treatment indicator) with $X$.}
\[
  Y_{it} \;=\; \alpha_i + \lambda_t
        \;+\; \delta_0\,(D_i \times \text{Post}_t)
        \;+\; \boldsymbol{\tau}\,(D_i \times \text{Post}_t \times X_i)
        \;+\; \boldsymbol{\beta}\,X_i
        \;+\; \varepsilon_{it}
\]
Now $\delta_0 + \boldsymbol{\tau}\,X_i$ is the treatment effect for a unit with covariate value $X_i$.\\[0.3em]
The ATT is recovered by averaging $\hat\delta_0 + \hat{\boldsymbol{\tau}}\,X_i$ over the \textit{treated} sample.

\vspace{0.3em}
\meta{Standard in heterogeneous-TE DiD (e.g.\ Callaway-Sant'Anna with covariates). In practice, use \texttt{drdid} or \texttt{csdid} rather than hand-coding this.}
\end{frame}

%% ---------- Slide: Both worries ----------
\begin{frame}{Both worries at once: the fully saturated spec}
\vspace{0.2em}
Include \textit{all three} interaction blocks:
\[
  Y_{it} \;=\; \alpha_i + \lambda_t
        \;+\; \underbrace{\delta_0\,(D_i \times \text{Post}_t)}_{\text{avg TE}}
        \;+\; \underbrace{\boldsymbol{\tau}\,(D_i \times \text{Post}_t \times X_i)}_{\text{TE varies in }X}
        \;+\; \underbrace{\boldsymbol{\gamma}\,(\text{Post}_t \times X_i)}_{\text{returns vary over time}}
        \;+\; \boldsymbol{\beta}\,X_i + \varepsilon_{it}
\]
\vspace{0.1em}
\textbf{Recovery of the ATT:}
\[
  \widehat{\text{ATT}} \;=\; \frac{1}{n_T}\sum_{i:\,D_i=1}
      \left(\hat\delta_0 + \hat{\boldsymbol{\tau}}\,X_i\right)
\]
\vspace{0.1em}
\highlight{This is Spec C in the lab do-file} (``Fully Saturated TWFE'').\\[0.2em]
\meta{Worries 1 and 2 together exhaust the ways an additive OLS spec can fail. The non-parametric fix is to leave OLS entirely and use OR, IPW, or DR.}
\end{frame}

%% PART 2 — SEMIPARAMETRIC ESTIMATORS
%% ====================================================================

%% ---------- Slide: IPW (Abadie 2005) ----------
\begin{frame}{IPW --- Abadie (2005): reweight the controls}
\vspace{0.2em}
\textbf{Intuition.} Weight the control group so it \textit{looks like the treated group} in $X$. Then the reweighted control trend is a valid counterfactual.

\vspace{0.3em}
\textbf{Estimator.}
\[
  \widehat{\text{ATT}}_{\text{IPW}}
  \;=\; \frac{1}{n}\sum_{i=1}^n
        \underbrace{\frac{D_i - \hat p(X_i)}{1 - \hat p(X_i)}}_{\text{IPW weight}} \cdot \Delta Y_i
        \;\Bigg/\; \bar{D}
\]
where $\hat p(X_i) = \widehat{\Pr}(D_i=1\mid X_i)$ is the \textbf{propensity score} and $\Delta Y_i = Y_{i,\text{post}} - Y_{i,\text{pre}}$.

\vspace{0.3em}
\textbf{Five steps.}
\begin{enumerate}\setlength\itemsep{0.15em}\small
  \item \highlight{Specify and estimate the propensity score.} Regress $D_i$ on $X_i$ (logit or probit).
  \item \highlight{Check overlap.} Plot $\hat p(X)$ by treatment status. Trim if needed.
  \item \highlight{Construct weights.} $w_i = (D_i - \hat p(X_i)) / (1-\hat p(X_i))$.
  \item \highlight{Take the weighted average.} $\widehat{\text{ATT}}_{\text{IPW}} = (\sum w_i \Delta Y_i) / n_T$.
  \item \highlight{Standard error.} Bootstrap or influence-function SE.
\end{enumerate}
\end{frame}

%% ---------- Slide: Overlap ----------
\begin{frame}{Step 2 is non-negotiable: check overlap}
\vspace{0.4em}
IPW weights blow up when $\hat p(X_i) \to 1$ for control units. \\[0.4em]

\begin{itemize}\setlength\itemsep{0.4em}
  \item \textbf{Common support:} $\hat p(X_i) \in [\epsilon,\, 1-\epsilon]$ for all $i$. If some treated units have $\hat p \approx 1$, no control unit resembles them --- you have no counterfactual.
  \item \textbf{Diagnosis:} histogram of $\hat p$ separately for $D=0$ and $D=1$. They should substantially overlap.
  \item \textbf{Fix:} trim units outside support (Crump et al.\ 2009: keep $\hat p \in [0.1, 0.9]$) or use a trimmed ATT.
\end{itemize}

\vspace{0.5em}
\emphbox{If the propensity score separates treated from controls perfectly, IPW is undefined. No method fixes lack of overlap --- it is a \textit{data} problem, not an estimation problem.}
\end{frame}

%% ---------- Slide: OR (HIT 1997) ----------
\begin{frame}{OR --- Heckman, Ichimura \& Todd (1997): impute the counterfactual}
\vspace{0.2em}
\textbf{Intuition.} Model the \textit{trend} of untreated potential outcomes conditional on $X$. Use that model to predict what the treated group \textit{would have done} absent treatment.

\vspace{0.3em}
\textbf{Estimator.}
\[
  \widehat{\text{ATT}}_{\text{OR}}
  \;=\; \frac{1}{n_T}\sum_{i:\,D_i=1}
        \Bigl[\Delta Y_i \;-\; \hat\mu_0(X_i)\Bigr]
\]
where $\hat\mu_0(X_i) = \widehat{E}[\Delta Y \mid D=0,\, X]$ is the \textbf{outcome model} --- the predicted trend for a control unit with characteristics $X_i$.

\vspace{0.3em}
\textbf{Steps.}
\begin{enumerate}\setlength\itemsep{0.15em}\small
  \item \highlight{First-difference.} Compute $\Delta Y_i = Y_{i,\text{post}} - Y_{i,\text{pre}}$ for all units.
  \item \highlight{Fit the outcome model.} Regress $\Delta Y$ on $X$ \textit{using controls only} ($D_i=0$).
  \item \highlight{Predict counterfactuals.} Evaluate $\hat\mu_0(X_i)$ for every treated unit.
  \item \highlight{Take the mean gap.} ATT $= \overline{\Delta Y}^T - \overline{\hat\mu_0(X)}^T$.
  \item \highlight{Standard error.} Bootstrap.
\end{enumerate}
\end{frame}

%% ---------- Slide: DR ----------
\begin{frame}{Doubly robust --- Sant'Anna \& Zhao (2020)}
\vspace{0.2em}
\textbf{Intuition.} Combine OR and IPW. One model can be misspecified and the estimator is still consistent --- but not both.

\vspace{0.3em}
\textbf{Estimator.}
\[
  \widehat{\text{ATT}}_{\text{DR}}
  \;=\; \frac{1}{\bar D}\,\frac{1}{n}\sum_{i=1}^n
        \underbrace{\frac{D_i - \hat p(X_i)}{1 - \hat p(X_i)}}_{\text{IPW correction}}
        \;\cdot\;
        \underbrace{\bigl[\Delta Y_i - \hat\mu_0(X_i)\bigr]}_{\text{OR residual}}
\]
\vspace{0.1em}
\begin{itemize}\setlength\itemsep{0.3em}
  \item If $\hat\mu_0$ is correct but $\hat p$ is wrong $\;\Rightarrow\;$ consistent. The OR residuals are mean-zero for the controls, so misweighting doesn't matter.
  \item If $\hat p$ is correct but $\hat\mu_0$ is wrong $\;\Rightarrow\;$ consistent. IPW balances $X$, so the misspecified trend washes out.
  \item If \textit{both} are wrong $\;\Rightarrow\;$ biased. ``Doubly robust'' $\neq$ doubly unbeatable.
\end{itemize}

\vspace{0.2em}
\meta{Implementation: \texttt{drdid re \$X, time(year) ivar(id) tr(ever\_treated) dripw} in Stata.}
\end{frame}

%% ---------- Slide: MC setup ----------
\begin{frame}{A Monte Carlo to settle the rankings}
\vspace{0.6em}
\begin{itemize}
  \item Compare four estimators on the same data: \textbf{TWFE} (with $X$ as additive controls), \textbf{OR}, \textbf{IPW}, \textbf{DR}.
  \item \textbf{4 DGPs} (Sant'Anna's design): vary whether the OR model and the pscore model are correctly specified.
  \item \textbf{Sample size:} 1{,}000 units. \;\textbf{Replications:} 10{,}000.
  \item \textbf{Report:} bias, RMSE, SE, 95\% coverage, CI length.
\end{itemize}
\vspace{0.5em}
\begin{center}
\small\color{warmgray}\itshape
The question we want answered: \;when does DR pay off, and when does it break?
\end{center}

%% ---------- Slide: MC takeaway ----------
\begin{frame}{What the Monte Carlo teaches}
\vspace{0.4em}
\begin{itemize}\small
  \item \textbf{TWFE with additive $X$:} never wins. \;Coverage is zero in every DGP. \highlight{Stop using it as a default.}
  \item \textbf{OR:} efficient when the outcome model is right. Disaster when it's wrong (Bias $-5$, coverage $1.5\%$).
  \item \textbf{IPW:} less efficient but \textit{robust to outcome misspecification}. The unsexy workhorse.
  \item \textbf{DR:} best of both \textit{when at least one is right}. \;Not better than IPW when \textit{both} are wrong.
\end{itemize}
\vspace{0.4em}
\begin{center}
\small\color{warmgray}\itshape
``Doubly robust'' means \textit{robust to one misspecification.} \;Not two. \;Not magic.
\end{center}
\end{frame}

%% ---------- Slide: Conclusion ----------
\begin{frame}{The lesson: specification is not cosmetic}
\vspace{0.3em}
All five estimators today require you to \textit{correctly specify something}:
\vspace{0.3em}
\begin{itemize}\setlength\itemsep{0.35em}
  \item \textbf{OLS specs (A, B, C):} you must commit to how $X$ enters $Y(0)$ \textit{and} whether $\delta$ is homogeneous in $X$. Wrong choice $\Rightarrow$ wrong ATT.
  \item \textbf{OR:} the outcome model $\hat\mu_0(X)$ must be correctly specified.
  \item \textbf{IPW:} the propensity score $\hat p(X)$ must be correctly specified, and overlap must hold.
  \item \textbf{DR:} \textit{at least one} of the two must be right.
\end{itemize}
\vspace{0.4em}
\emphbox{Adding covariates does not make your DiD credible. \;It transfers the credibility question from ``does PT hold?'' to ``is my model of $Y(0)$ or $p(X)$ correctly specified?'' \;You still have to answer it.}
\end{frame}
